"use client";

import { useRouter, useSearchParams } from "next/navigation";

type FilterOption = {
  value: string;
  label: string;
};

type EventFiltersProps = {
  cities: FilterOption[];
  categories: FilterOption[];
  selectedCity: string;
  selectedCategory: string;
  selectedWhen: string;
};

export default function EventFilters({
  cities,
  categories,
  selectedCity,
  selectedCategory,
  selectedWhen,
}: EventFiltersProps) {
  const router = useRouter();
  const searchParams = useSearchParams();

  function updateFilter(
    key: "city" | "category" | "when",
    value: string
  ) {
    const params = new URLSearchParams(searchParams.toString());

    if (value) {
      params.set(key, value);
    } else {
      params.delete(key);
    }

    router.push(
      params.toString()
        ? `/events?${params.toString()}`
        : "/events"
    );
  }

  function clearFilters() {
    const params = new URLSearchParams(searchParams.toString());

    params.delete("city");
    params.delete("category");
    params.delete("when");

    router.push(
      params.toString()
        ? `/events?${params.toString()}`
        : "/events"
    );
  }

  const hasFilters =
    Boolean(selectedCity) ||
    Boolean(selectedCategory) ||
    Boolean(selectedWhen);

  return (
    <section className="border-b border-slate-200 bg-white">
      <div className="mx-auto max-w-7xl px-6 py-6">

        <div className="grid gap-4 md:grid-cols-3">

          {/* CITY */}
          <div>
            <label
              htmlFor="event-city"
              className="mb-2 block text-xs font-bold uppercase tracking-wide text-slate-400"
            >
              City
            </label>

            <select
              id="event-city"
              value={selectedCity}
              onChange={(event) =>
                updateFilter("city", event.target.value)
              }
              className="w-full rounded-xl border border-slate-200 bg-slate-50 px-4 py-3 font-semibold text-slate-800 outline-none transition focus:border-orange-500 focus:ring-2 focus:ring-orange-100"
            >
              <option value="">All Cities</option>

              {cities.map((city) => (
                <option
                  key={city.value}
                  value={city.value}
                >
                  {city.label}
                </option>
              ))}
            </select>
          </div>

          {/* CATEGORY */}
          <div>
            <label
              htmlFor="event-category"
              className="mb-2 block text-xs font-bold uppercase tracking-wide text-slate-400"
            >
              Category
            </label>

            <select
              id="event-category"
              value={selectedCategory}
              onChange={(event) =>
                updateFilter(
                  "category",
                  event.target.value
                )
              }
              className="w-full rounded-xl border border-slate-200 bg-slate-50 px-4 py-3 font-semibold text-slate-800 outline-none transition focus:border-orange-500 focus:ring-2 focus:ring-orange-100"
            >
              <option value="">All Events</option>

              {categories.map((category) => (
                <option
                  key={category.value}
                  value={category.value}
                >
                  {category.label}
                </option>
              ))}
            </select>
          </div>

          {/* WHEN */}
          <div>
            <label
              htmlFor="event-when"
              className="mb-2 block text-xs font-bold uppercase tracking-wide text-slate-400"
            >
              When
            </label>

            <select
              id="event-when"
              value={selectedWhen}
              onChange={(event) =>
                updateFilter(
                  "when",
                  event.target.value
                )
              }
              className="w-full rounded-xl border border-slate-200 bg-slate-50 px-4 py-3 font-semibold text-slate-800 outline-none transition focus:border-orange-500 focus:ring-2 focus:ring-orange-100"
            >
              <option value="upcoming">Upcoming</option>
              <option value="today">Today</option>
              <option value="weekend">This Weekend</option>
              <option value="week">This Week</option>
              <option value="month">This Month</option>
            </select>
          </div>

        </div>

        {/* ACTIVE FILTERS */}
        {hasFilters && (
          <div className="mt-5 flex flex-wrap items-center gap-3">

            <span className="text-xs font-bold uppercase tracking-wide text-slate-400">
              Filters applied
            </span>

            {selectedCity && (
              <button
                type="button"
                onClick={() =>
                  updateFilter("city", "")
                }
                className="rounded-full bg-orange-50 px-4 py-2 text-sm font-bold text-orange-700 transition hover:bg-orange-100"
              >
                {cities.find(
                  (city) => city.value === selectedCity
                )?.label || selectedCity}
                <span className="ml-2">×</span>
              </button>
            )}

            {selectedCategory && (
              <button
                type="button"
                onClick={() =>
                  updateFilter("category", "")
                }
                className="rounded-full bg-orange-50 px-4 py-2 text-sm font-bold text-orange-700 transition hover:bg-orange-100"
              >
                {categories.find(
                  (category) =>
                    category.value === selectedCategory
                )?.label || selectedCategory}
                <span className="ml-2">×</span>
              </button>
            )}

            {selectedWhen &&
              selectedWhen !== "upcoming" && (
                <button
                  type="button"
                  onClick={() =>
                    updateFilter("when", "upcoming")
                  }
                  className="rounded-full bg-orange-50 px-4 py-2 text-sm font-bold text-orange-700 transition hover:bg-orange-100"
                >
                  {selectedWhen === "today"
                    ? "Today"
                    : selectedWhen === "weekend"
                      ? "This Weekend"
                      : selectedWhen === "week"
                        ? "This Week"
                        : selectedWhen === "month"
                          ? "This Month"
                          : selectedWhen}
                  <span className="ml-2">×</span>
                </button>
              )}

            <button
              type="button"
              onClick={clearFilters}
              className="rounded-full px-4 py-2 text-sm font-bold text-slate-500 underline underline-offset-2 hover:text-slate-900"
            >
              Clear filters
            </button>

          </div>
        )}

      </div>
    </section>
  );
}
